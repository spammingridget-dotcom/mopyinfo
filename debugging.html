Advanced Event Watcher & Trace Logger

```lua
local Debugger = {}
Debugger.__index = Debugger

function Debugger.new()
    return setmetatable({ traces = {}, listening = {} }, Debugger)
end

function Debugger:Watch(obj, event)
    local s = obj[event]
    if typeof(s) ~= "RBXScriptSignal" then return end

    local c = s:Connect(function(...)
        table.insert(self.traces, {
            time = os.clock(),
            source = obj.Name .. "." .. event,
            args = {...}
        })
    end)

    table.insert(self.listening, c)
end

function Debugger:Dump()
    for _, t in ipairs(self.traces) do
        warn("[trace]", t.time, t.source, unpack(t.args))
    end
end

local dbg = Debugger.new()

dbg:Watch(workspace, "ChildAdded")
dbg:Watch(game.Players, "PlayerAdded")

dbg:Dump()
```
