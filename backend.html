Advanced Datastore Wrapper

```lua
local DS = game:GetService("DataStoreService"):GetDataStore("V1")

local PlayerData = {}
PlayerData.__index = PlayerData

local function retry(fn)
    for i = 1, 5 do
        local ok, res = pcall(fn)
        if ok then return res end
        task.wait(.2 * i)
    end
    warn("datastore failed")
end

function PlayerData.new(id)
    return setmetatable({ id = id, cache = {}, loaded = false }, PlayerData)
end

function PlayerData:Load()
    self.cache = retry(function()
        return DS:GetAsync(self.id)
    end) or {}
    self.loaded = true
end

function PlayerData:Set(k,v)
    self.cache[k] = v
end

function PlayerData:Save()
    if not self.loaded then return end
    retry(function()
        DS:SetAsync(self.id, self.cache)
    end)
end

local store = {}

game.Players.PlayerAdded:Connect(function(plr)
    local pd = PlayerData.new(plr.UserId)
    store[plr] = pd
    pd:Load()
end)

game.Players.PlayerRemoving:Connect(function(plr)
    store[plr]:Save()
end)
```
