Advanced UI Controller Script

```lua
local Signal = {}
Signal.__index = Signal
function Signal.new()
    return setmetatable({ _binds = {} }, Signal)
end
function Signal:Connect(fn)
    table.insert(self._binds, fn)
end
function Signal:Fire(...)
    for _, fn in ipairs(self._binds) do fn(...) end
end

local UIController = {}
UIController.__index = UIController

function UIController.new(root)
    return setmetatable({
        root = root,
        state = false,
        changed = Signal.new()
    }, UIController)
end

function UIController:Set(state)
    self.state = state
    self.changed:Fire(state)
end

local obj = UIController.new(script.Parent)

obj.changed:Connect(function(active)
    obj.root.Frame.BackgroundTransparency = active and .15 or .55
end)

obj.root.Frame.MouseButton1Click:Connect(function()
    obj:Set(not obj.state)
end)
```
